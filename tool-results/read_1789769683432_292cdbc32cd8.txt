    84→const fmtInt = (n: number) => n.toLocaleString('fr-FR')
    85→const pct = (num: number, denom: number) =>
    86→  denom > 0 ? (num / denom) * 100 : 0
    87→
    88→function fmtDay(iso: string): string {
    89→  const d = new Date(iso)
    90→  if (isNaN(d.getTime())) return iso
    91→  return d.toLocaleDateString('fr-FR', { day: '2-digit', month: '2-digit' })
    92→}
    93→
    94→function fmtDayMonth(iso: string): string {
    95→  const d = new Date(iso)
    96→  if (isNaN(d.getTime())) return iso
    97→  return d.toLocaleDateString('fr-FR', { day: 'numeric', month: 'short' })
    98→}
    99→
   100→/** Deterministic mock delta so the same metric always shows the same %.
   101→ *  (avoids the number flickering on every render). */
   102→function mockDelta(seed: string): number {
   103→  let h = 0
   104→  for (let i = 0; i < seed.length; i++) h = (h * 31 + seed.charCodeAt(i)) >>> 0
   105→  return 3 + (h % 22)
   106→}

   131→/* --------------------------------- KPI card -------------------------------- */
   132→
   133→interface KpiCardProps {
   134→  icon: React.ComponentType<{ className?: string }>
   135→  label: string
   136→  value: number | string
   137→  delta?: number
   138→  sublabel?: string
   139→  accent?: string
   140→}
   141→
   142→function KpiCard({ icon: Icon, label, value, delta, sublabel, accent }: KpiCardProps) {
   143→  return (
   144→    <Card className="overflow-hidden">
   145→      <CardHeader className="pb-2">
   146→        <div className="flex items-center justify-between">
   147→          <CardDescription className="text-xs font-medium uppercase tracking-wide">
   148→            {label}
   149→          </CardDescription>
   150→          <span
   151→            className="flex size-8 items-center justify-center rounded-md"
   152→            style={{ backgroundColor: accent ?? 'var(--primary)' + '20', color: accent ?? 'var(--primary)' }}
   153→          >
   154→            <Icon className="size-4" />
   155→          </span>
   156→        </div>
   157→      </CardHeader>
   158→      <CardContent className="pt-0">
   159→        <div className="flex items-baseline gap-2">
   160→          <span className="text-2xl font-bold tracking-tight tabular-nums">
   161→            {typeof value === 'number' ? fmtInt(value) : value}
   162→          </span>
   163→          {delta !== undefined && delta > 0 && (
   164→            <span className="flex items-center gap-0.5 text-xs font-medium text-emerald-600 dark:text-emerald-400">
   165→              <ArrowUpRight className="size-3" />
   166→              +{delta.toFixed(1)}%
   167→            </span>
   168→          )}
   169→        </div>
   170→        {sublabel && (
   171→          <p className="mt-1 text-xs text-muted-foreground">{sublabel}</p>
   172→        )}
   173→      </CardContent>
   174→    </Card>
   175→  )
   176→}

   566→            <div className="flex flex-col items-center justify-center py-8 text-center">
   567→              <Mail className="mb-2 size-8 text-muted-foreground/50" />
   568→              <p className="text-sm text-muted-foreground">Aucune campagne pour le moment.</p>
   569→              <Button size="sm" className="mt-3" onClick={() => setView('campaign-new')}>
   570→                <Plus className="size-4" />
   571→                Créer une campagne
   572→              </Button>
   573→            </div>
   574→          ) : (
   575→            <div className="grid gap-3 md:grid-cols-3">
   576→              {recentCampaigns.map((c) => {
   577→                const cOpen = pct(c.opened, c.delivered || c.sent || 1)
   578→                const cDeliv = pct(c.delivered, c.sent || 1)
   579→                return (
   580→                  <button
   581→                    key={c.id}
   582→                    type="button"
   583→                    onClick={() => setView('campaign-detail', c.id)}
   584→                    className="group flex flex-col gap-2 rounded-lg border border-border bg-card p-4 text-left transition-colors hover:border-primary/40 hover:bg-accent/30"
   585→                  >
   586→                    <div className="flex items-start justify-between gap-2">
   587→                      <span className="line-clamp-1 text-sm font-semibold">
   588→                        {c.name}
   589→                      </span>
   590→                      {statusBadge(c.status)}
   591→                    </div>
   592→                    <p className="line-clamp-1 text-xs text-muted-foreground">
   593→                      {c.subject || '—'}
   594→                    </p>
   595→                    <div className="mt-1 grid grid-cols-3 gap-1 text-xs">
   596→                      <div>
   597→                        <div className="font-medium tabular-nums">{fmtInt(c.sent)}</div>
   598→                        <div className="text-[10px] text-muted-foreground">Envoyés</div>
   599→                      </div>
   600→                      <div>
   601→                        <div className="font-medium tabular-nums">{cDeliv.toFixed(0)}%</div>
   602→                        <div className="text-[10px] text-muted-foreground">Délivrés</div>
   603→                      </div>
   604→                      <div>
   605→                        <div className="font-medium tabular-nums text-primary">{cOpen.toFixed(0)}%</div>
   606→                        <div className="text-[10px] text-muted-foreground">Ouverts</div>
   607→                      </div>
   608→                    </div>
   609→                  </button>
   610→                )
   611→              })}
   612→            </div>
   613→          )}
   614→        </CardContent>
   615→      </Card>
   616→
   617→      {/* Quick actions */}
   618→      <Card>
   619→        <CardHeader>
   620→          <CardTitle className="text-base">Actions rapides</CardTitle>
   621→          <CardDescription>Continuez là où vous vous êtes arrêté</CardDescription>
   622→        </CardHeader>
   623→        <CardContent>
   624→          <div className="grid gap-3 sm:grid-cols-2 lg:grid-cols-4">
   625→            <QuickAction
   626→              icon={Plus}
   627→              label="Nouvelle campagne"
   628→              description="Rédigez et envoyez"
   629→              onClick={() => setView('campaign-new')}
   630→            />
   631→            <QuickAction
   632→              icon={Upload}
   633→              label="Importer contacts"
   634→              description="CSV ou intégration"
   635→              onClick={() => setView('contacts')}
   636→            />
   637→            <QuickAction
   638→              icon={Globe}
   639→              label="Configurer le domaine"
   640→              description="Authentifiez votre envoi"
   641→              onClick={() => setView('domain')}
   642→            />
   643→            <QuickAction
   644→              icon={BarChart3}
   645→              label="Voir statistiques"
   646→              description="Rapports détaillés"
   647→              onClick={() => setView('stats')}
   648→            />
   649→          </div>
   650→        </CardContent>
   651→      </Card>
   652→    </div>
   653→  )
   654→}
